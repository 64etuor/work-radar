# Control Room Live Layered Schematic — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an always-on, polling, per-project layered Control Room diagram — runtime INFRA bar + READINESS gate strip + work-pipeline lanes — that renders even at zero work, driven by a read-only extension of the snapshot fold.

**Architecture:** The governance snapshot fold (`boule snapshot` CLI, consumed by the T4b `control_room.rs` runner) gains `infra` + `readiness` blocks, computed by the SAME `computeRepoReadiness` helper the `boule tick` path uses (so the dashboard verdict equals what a real tick would gate on). The Rust `ControlRoomSnapshot` grows two typed fields; the frontend polls it (~4s) and renders three layers.

**Tech Stack:** TypeScript (governance fold, Node `--experimental-strip-types`), Rust (Tauri/specta IPC), React + TanStack Query + shadcn (frontend). Tests: vitest (governance + frontend), `cargo test` (Rust).

**Spec:** `docs/superpowers/specs/2026-06-09-control-room-live-schematic-design.md`

**Conventions:** Commands are canonical in `docs/reference/commands.md`. Run `cargo` from `src-tauri/` with `$HOME/.cargo/bin` on PATH. Regenerate `src/bindings.ts` via `cargo test` (drift gate). Governance tests: `cd governance && pnpm vitest run`. Never hand-edit `bindings.ts`. INVARIANT #1: this whole path is read-only (no event writes).

```ts
export function evaluateReadiness(input: ReadinessInput): ReadinessReport {
  const preflight = evaluatePreflight(input.signals);
  const env = input.signals.env;

  const gates: ReadinessGates = {
    preflight: preflight.ready,
    conductor: parseConductorMode(env.BOULE_CONDUCTOR_MODE),
    observer: parseObserverMode(env.BOULE_OBSERVER_MODE),
    auto: parseAutoMode(env.BOULE_AUTO_MODE) === "type2",
    stall: input.stalledCount === 0,
  };

  const blockers: string[] = [];
  if (!gates.preflight)
    blockers.push("preflight infra not ready (cmux socket/binary or git worktree)");
  if (!gates.conductor) blockers.push("BOULE_CONDUCTOR_MODE!=live — no real sessions would spawn");
  if (!gates.observer)
    blockers.push("BOULE_OBSERVER_MODE!=live — crashes/stalls undetected (unsafe unattended)");
  if (!gates.auto) blockers.push("BOULE_AUTO_MODE!=type2 — no unattended step advance");
  if (!gates.stall)
    blockers.push(`${input.stalledCount} stalled task(s) — resolve before unattended dispatch`);

  const warnings: string[] = [];
  const autonomy = parseAutonomyLevel(env.BOULE_AUTONOMY_LEVEL);
  if (autonomy === "auto_plan" || autonomy === "full")
    warnings.push(
      `BOULE_AUTONOMY_LEVEL=${autonomy} — INVARIANT #1 OPENED: AI auto-accepts ${autonomy === "full" ? "plan + spec" : "plan"} (test/dev only; Type-1 decisions + stalls still halt for a human)`,
    );

  return { ready: blockers.length === 0, blockers, warnings, gates, preflight };
}
```

(Keep the existing `parseConductorMode`/`parseObserverMode`/`parseAutoMode`/`parseAutonomyLevel` imports — they already exist in the file.)

```ts
import { spawnSync } from "node:child_process";

import { parseConductorConfig, type ConductorConfig } from "../mcp/config.ts";
import { listStalledTasks } from "../orchestrator/stall.ts";
import { loadPolicyOrStrict, repoEnv } from "../repo/configIo.ts";
import { boulePaths } from "../repo/paths.ts";
import type { ResolvedPolicy } from "../repo/policy.ts";
import { evaluateReadiness, type ReadinessReport } from "./readiness.ts";

export interface RepoReadiness {
  policy: ResolvedPolicy;
  env: Record<string, string | undefined>;
  cfg: ConductorConfig;
  stalledCount: number;
  readiness: ReadinessReport;
  /** "ok" | "fallback" — config load posture (callers may warn on fallback). */
  source: string;
  reason?: string;
}

/** Gather the per-repo readiness EXACTLY as `boule tick` does: policy-derived env (autonomyLevel →
 *  autoMode, etc.) + cheap cmux/git probes + stall count → evaluateReadiness. One-shot, read-only,
 *  no retry (INVARIANT #5). Used by BOTH `cmdTick` and `cmdSnapshot` so the Control Room verdict
 *  always equals what a real tick would gate on. */
export async function computeRepoReadiness(repoRoot: string): Promise<RepoReadiness> {
  const loaded = await loadPolicyOrStrict(repoRoot);
  const env = { ...process.env, ...repoEnv(repoRoot, loaded.policy) };
  const cfg = parseConductorConfig(env);
  const probe = (cmd: string, args: string[]): boolean => {
    try {
      return spawnSync(cmd, args, { encoding: "utf8" }).status === 0;
    } catch {
      return false;
    }
  };
  const stalledCount = (await listStalledTasks(boulePaths(repoRoot).workspace)).length;
  const readiness = evaluateReadiness({
    signals: {
      env,
      cmuxBinaryOk: probe(cfg.cmuxCmd, ["--version"]),
      gitWorktreeOk: probe("git", ["-C", repoRoot, "worktree", "list"]),
    },
    stalledCount,
  });
  return {
    policy: loaded.policy,
    env,
    cfg,
    stalledCount,
    readiness,
    source: loaded.source,
    reason: loaded.reason,
  };
}
```

```ts
import type { PreflightCheck, ReadinessGates } from "../bindings";

export type InfraTone = "ok" | "down" | "advisory";

/** ok → green, failing-required → red (down), failing-advisory → amber. */
export function infraTone(check: PreflightCheck): InfraTone {
  if (check.ok) return "ok";
  return check.level === "required" ? "down" : "advisory";
}

/** Armed ⇔ every readiness gate passes (mirrors readiness.ready, derived from gates so the UI can
 *  render the verdict without a second round-trip). */
export function isArmed(gates: ReadinessGates): boolean {
  return gates.preflight && gates.conductor && gates.observer && gates.auto && gates.stall;
}
```

```tsx
import { AlertTriangle, Check, ShieldAlert, ShieldCheck, X } from "lucide-react";

import type { ReadinessView } from "../../bindings";
import { isArmed } from "../../lib/controlRoom";
import { Badge } from "../ui/badge";

const GATE_LABELS: { key: keyof ReadinessView["gates"]; label: string }[] = [
  { key: "preflight", label: "preflight" },
  { key: "conductor", label: "conductor" },
  { key: "observer", label: "observer" },
  { key: "auto", label: "auto" },
  { key: "stall", label: "stall" },
];

export function ReadinessStrip({ readiness }: { readiness: ReadinessView }) {
  const armed = isArmed(readiness.gates);
  return (
    <div className="cr-readiness" data-armed={armed || undefined}>
      <div className="cr-readiness-gates">
        {GATE_LABELS.map(({ key, label }) => {
          const pass = readiness.gates[key];
          return (
            <span key={key} className="cr-gate" data-pass={pass || undefined}>
              {pass ? <Check size={12} aria-hidden /> : <X size={12} aria-hidden />}
              {label}
            </span>
          );
        })}
        <Badge variant={armed ? "secondary" : "destructive"} className="cr-readiness-verdict">
          {armed ? <ShieldCheck size={12} aria-hidden /> : <ShieldAlert size={12} aria-hidden />}
          {armed ? "ARMED" : "NOT ARMED"}
        </Badge>
      </div>
      {!armed && readiness.blockers.length > 0 && (
        <ul className="cr-readiness-blockers">
          {readiness.blockers.map((b) => (
            <li key={b}>{b}</li>
          ))}
        </ul>
      )}
      {readiness.warnings.map((w) => (
        <div key={w} className="cr-readiness-warning">
          <AlertTriangle size={12} aria-hidden /> {w}
        </div>
```
